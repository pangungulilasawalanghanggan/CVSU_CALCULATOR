import Link from "next/link";
import { cookies } from "next/headers";
import { redirect } from "next/navigation";
import { demoEnabled, loginDestination, readSession, SESSION_COOKIE } from "../../lib/auth";
import DemoSignInButton from "../../components/DemoSignInButton";
export const dynamic = "force-dynamic";
export default function LoginPage({ searchParams }: { searchParams: { next?: string } }) {
    const destination = loginDestination(searchParams.next);
    if (readSession(cookies().get(SESSION_COOKIE)?.value)) redirect(destination);
    const configured = demoEnabled();
    return <div className="hero-wash min-h-[70vh] px-4 py-16"><div className="card mx-auto max-w-md p-8 sm:p-10">
      <p className="section-label mb-3">Localhost study demo</p>
      <h1 className="text-3xl font-extrabold text-forest-900">Try the calculators</h1>
      <p className="mt-4 text-ink-soft">Open the sample student account with one click. All calculators and the grading guide use the same design as the main app.</p>
      <DemoSignInButton configured={configured} destination={destination}/>
      <p className="mt-5 text-xs text-ink-soft">Simulated sign-in only. No Google account, password, registration, or OAuth setup is needed. Demo Student and demo@cvsu.edu.ph are sample details.</p>
      {!configured && <p role="alert" className="mt-4 text-sm text-red-700">This demo is disabled in production. Run npm run dev on localhost.</p>}
      <Link href="/" className="mt-6 inline-block text-sm font-semibold text-forest-800 hover:underline">← Back to home</Link>
    </div></div>;
}
